\newcommand{\card}[6]{%

    \fill[#3!13] (#1,#2) rectangle ++(\CW,-\HH);
  \end{scope}

  \node[text width=\CW cm-6pt,align=center,font=\sffamily\bfseries\scriptsize,inner sep=0pt]
    at ($(#1,#2)+(0.5*\CW,-0.5*\HH)$) {#4};
  \node[anchor=north west,text width=\CW cm-8pt,align=left,font=\sffamily\scriptsize,inner sep=0pt]
    at ($(#1,#2)+(0.14,-\HH-0.14)$) {#5};
  \node[anchor=south,inner sep=0pt] at ($(#1,#2)+(0.5*\CW,-\HH-\BH+0.12)$) {#6};
}
